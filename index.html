Project 1 - Zig Example

This is an overview of garbage with some Zig code:


const std = @import("std");
const time = std.time.milliTimestamp;
const StOutput = std.io.getStdOut().writer();
const g: i32 = 1_000_000;
var bw = std.io.bufferedWriter(StOutput);
const writer = bw.writer();

pub fn main() !void
{
    const start_time = time();

    var i: usize = 0;
    while (i <= g) : (i += 1)
    {
        try writer.print("{s}{}\n", .{".ic ",i});
    }

    try bw.flush();

    const end_time = time();
    const dur: f64 = @as(f64, @floatFromInt(end_time - start_time)) / 1000;
    try writer.print("{d}s {}ms {}ms\n", .{ dur, start_time, end_time });
    try bw.flush();
}

This is an overview of a firewall checker quicktool:


import subprocess

def check_firewall_rules():
    result = subprocess.run(['ufw', 'status'], stdout=subprocess.PIPE)
    if 'ALLOW' in result.stdout.decode():
        print("Firewall rules are in place.")
    else:
        print("No firewall rules found.")

if __name__ == "__main__":
    check_firewall_rules()
